& {
    $ErrorActionPreference = 'Stop'
    $RunPairedCapture = $false
    $VolumeSymbol = ''  # Exact active TWS symbol for the optional paired capture
    $SdkReceipt = ''  # Optional exact retained laptop sdk-environment.json
    $Pin = '1ea5672e0367011f83745e644f876e363cb77b76'
    $Tree = '4725325f1d0adb6c9a0bda4d6f42d3421eb9a0cc'
    $Repo = Read-Host 'Existing ibkr-trading-system checkout folder'
    $Base = 'C:\tmp'
    $status = 'WINDOWS_POWERSHELL_51_REQUIRED'
    $captureRoot = $null
    $probeCode = $null
    $stage = 'PREPARED_ONLY'
    try {
        if ($PSVersionTable.PSEdition -ne 'Desktop' -or
            $PSVersionTable.PSVersion.Major -ne 5 -or
            $PSVersionTable.PSVersion.Minor -ne 1) { return }

        if ($RunPairedCapture -and $VolumeSymbol -cnotmatch '^[A-Z][A-Z0-9. -]{0,14}\z') {
            $status = 'PAIRED_CAPTURE_SYMBOL_REQUIRED'
            return
        }

        $status = 'SOURCE_CHECKOUT_UNAVAILABLE'
        if (!(Test-Path -LiteralPath $Repo -PathType Container)) { return }
        $Repo = (Get-Item -LiteralPath $Repo).FullName

        $status = 'SDK_RECEIPT_UNAVAILABLE'
        $paths = if ($SdkReceipt) { @($SdkReceipt) } else {
            Get-ChildItem -LiteralPath $Base -Directory | ForEach-Object {
                Join-Path $_.FullName 'sdk-environment.json'
            }
        }
        $receipts = @(foreach ($path in $paths) {
            if (!(Test-Path -LiteralPath $path -PathType Leaf)) { continue }
            try { $r = Get-Content -LiteralPath $path -Raw | ConvertFrom-Json }
            catch { continue }
            $receiptRoot = Split-Path -Parent (Get-Item -LiteralPath $path).FullName
            if ($r.schema -ceq 'IBKR_OFFICIAL_SDK_ENVIRONMENT_V1' -and
                $r.status -ceq 'SDK_ENVIRONMENT_PREPARED' -and
                $r.sdk_version -ceq '10.50.2' -and $r.protobuf_version -ceq '5.29.5' -and
                $r.sdk_profile_verified -is [bool] -and $r.sdk_profile_verified -and
                $r.source_count -eq 235 -and
                $r.archive_sha256 -ceq '673129e5cba58c4d77bc40647265f84ea42f605eccf88fa4c1221d62d12454f3' -and
                $r.source_sha256 -ceq '72dccfacba1295b3e4c45d5c97cf160cdb2e86d95b92de2ff408610f570b208b' -and
                $r.root -ieq $receiptRoot -and
                $r.python -ieq (Join-Path $receiptRoot 'venv\Scripts\python.exe') -and
                (Test-Path -LiteralPath $r.python -PathType Leaf)) { $r }
        })
        if ($receipts.Count -eq 0) { return }
        if ($receipts.Count -ne 1) { $status = 'SDK_RECEIPT_AMBIGUOUS'; return }
        $python = $receipts[0].python

        $status = 'PINNED_SOURCE_UNAVAILABLE'
        $git = (Get-Command git.exe -CommandType Application).Source
        function Invoke-PinnedGit([string[]]$GitArgs) {
            # Native progress can use stderr even when Git exits successfully.
            $ErrorActionPreference = 'Continue'
            $global:LASTEXITCODE = $null
            $lines = @(& $git @GitArgs 2>$null)
            [pscustomobject]@{ ExitCode = $global:LASTEXITCODE; Lines = $lines }
        }

        $status = 'WORKTREE_INVENTORY_UNAVAILABLE'
        $inventory = Invoke-PinnedGit -GitArgs @('-C', $Repo, 'worktree', 'list', '--porcelain')
        if ($inventory.ExitCode -ne 0) { return }

        $checkout = $null
        $pinnedWorktreeNotClean = $false
        foreach ($line in $inventory.Lines) {
            if (!$line.StartsWith('worktree ')) { continue }
            $candidate = $line.Substring(9).Replace('/', '\')
            if ($candidate -notmatch '^C:\\tmp\\ross-(source-check|readonly)-[0-9a-f]{32}$' -or
                !(Test-Path -LiteralPath $candidate -PathType Container)) { continue }
            $head = Invoke-PinnedGit -GitArgs @('-C', $candidate, 'rev-parse', 'HEAD')
            if ($head.ExitCode -ne 0 -or ($head.Lines -join '') -cne $Pin) { continue }
            $clean = Invoke-PinnedGit -GitArgs @('-C', $candidate, 'status', '--porcelain=v1', '--untracked-files=all')
            if ($clean.ExitCode -ne 0 -or $clean.Lines.Count) {
                $pinnedWorktreeNotClean = $true
                continue
            }
            $checkout = $candidate
            break
        }

        if (!$checkout -and $pinnedWorktreeNotClean) {
            $status = 'PINNED_WORKTREE_NOT_CLEAN'
            return
        }
        if (!$checkout) {
            $status = 'PINNED_SOURCE_UNAVAILABLE'
            $pinCheck = Invoke-PinnedGit -GitArgs @('-C', $Repo, 'cat-file', '-e', ($Pin + '^{commit}'))
            if ($pinCheck.ExitCode -ne 0) {
                $fetch = Invoke-PinnedGit -GitArgs @('-C', $Repo, 'fetch', '--quiet', 'origin', $Pin)
                if ($fetch.ExitCode -ne 0) { return }
            }
        }
        $status = 'PINNED_TREE_UNVERIFIED'
        $treeSource = if ($checkout) { $checkout } else { $Repo }
        $actualTree = Invoke-PinnedGit -GitArgs @('-C', $treeSource, 'rev-parse', ($Pin + '^{tree}'))
        if ($actualTree.ExitCode -ne 0 -or ($actualTree.Lines -join '') -cne $Tree) { return }

        if (!$checkout) {
            $status = 'PINNED_WORKTREE_CREATION_FAILED'
            $checkout = Join-Path $Base ('ross-readonly-' + [guid]::NewGuid().ToString('N'))
            $added = Invoke-PinnedGit -GitArgs @('-C', $Repo, '-c', 'core.longpaths=true', 'worktree', 'add', '--detach', $checkout, $Pin)
            if ($added.ExitCode -ne 0) { return }
        }
        $status = 'PINNED_WORKTREE_HEAD_UNVERIFIED'
        $actual = Invoke-PinnedGit -GitArgs @('-C', $checkout, 'rev-parse', 'HEAD')
        if ($actual.ExitCode -ne 0 -or ($actual.Lines -join '') -cne $Pin) { return }
        $status = 'PINNED_WORKTREE_NOT_CLEAN'
        $dirty = Invoke-PinnedGit -GitArgs @('-C', $checkout, 'status', '--porcelain=v1', '--untracked-files=all')
        if ($dirty.ExitCode -ne 0 -or $dirty.Lines.Count) { return }

        $status = 'SDK_PROFILE_UNVERIFIED'
        $attest = @'
import sys
sys.path.insert(0, sys.argv[1])
from src.ibkr.sdk_profile import require_supported_sdk
raise SystemExit(0 if require_supported_sdk() == '10.50.2' else 2)
'@
        $attestCode = & {
            $ErrorActionPreference = 'Continue'
            $global:LASTEXITCODE = $null
            $attest | & $python -I -B - $checkout *> $null
            $global:LASTEXITCODE
        }
        if ($attestCode -ne 0) { return }
        $status = 'SDK_10_50_2_VERIFIED_OFFLINE'
        if (!$RunPairedCapture) { return }

        $status = 'CAPTURE_STATUS_UNAVAILABLE'
        $stage = 'READ_ONLY_CAPTURE_INVOKED'
        $capture = Join-Path $checkout 'scripts\certification\capture_tws_volume_display.ps1'
        $captureArgs = @{
            ExpectedCommit = $Pin
            ExpectedTree = $Tree
            VolumeSymbol = $VolumeSymbol
            PythonExecutable = $python
        }
        $captureLines = New-Object 'System.Collections.Generic.List[string]'
        try { & $capture @captureArgs *>&1 | ForEach-Object { $captureLines.Add([string]$_) } }
        catch { }
        $roots = @($captureLines | Where-Object {
            $_ -cmatch '^Private automatic capture: C:\\tmp\\tws-volume-[0-9a-f]{8}$'
        } | Select-Object -Unique)
        if ($roots.Count -ne 1) { return }
        $captureRoot = $roots[0].Substring('Private automatic capture: '.Length)
        $receiptPath = Join-Path $captureRoot 'capture-status.json'
        if (!(Test-Path -LiteralPath $receiptPath -PathType Leaf)) { return }
        $result = Get-Content -LiteralPath $receiptPath -Raw | ConvertFrom-Json
        if ($result.schema -ceq 'ROSS_TWS_VOLUME_DISPLAY_CAPTURE_V1' -and
            $result.source -ceq $Pin -and
            $result.status -cin @('DISPLAY_CAPTURE_COMPLETE_UNITS_UNVERIFIED','COMPARISON_NOT_VERIFIED')) {
            $status = $result.status
            if ($result.probe_failure_code -cin @(
                'DISPLAY_RECORDER_NOT_READY_NO_PROBE_STARTED',
                'PAPER_API_LISTENER_QUERY_UNAVAILABLE_NO_NEW_REQUEST',
                'PAPER_API_LISTENER_CHANGED_NO_PROBE_STARTED',
                'PAPER_API_LISTENER_CHANGED_DURING_CAPTURE',
                'PROBE_TIMED_OUT','PROBE_DIAGNOSTIC_INCOMPLETE',
                'PROBE_DIAGNOSTIC_FAILED','PROBE_COMMAND_FAILED')) {
                $probeCode = $result.probe_failure_code
            }
        }
    } catch {
        # Use the fixed stage code. Exception text can contain local paths.
    } finally {
        if ($status -ceq 'SDK_10_50_2_VERIFIED_OFFLINE') {

            Write-Host ('Private pinned checkout ' + $checkout)
        }
        if ($captureRoot) { Write-Host ('Private capture directory ' + $captureRoot) }
        [ordered]@{ status=$status; stage=$stage; probe_failure_code=$probeCode; PAPER_READY=$false } |
            ConvertTo-Json -Compress
    }
